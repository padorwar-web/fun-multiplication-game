    <div class="game-container">

        <div class="stats-grid">
            <div class="stat-box">⭐ Score: <span id="score">0</span></div>
            <div class="stat-box">⏳ Time: <span id="time-text">30</span>s</div>
        </div>
        <div class="timer-bar-container">
            <div id="timer-bar" class="timer-bar"></div>
        </div>
        <div id="flashcard" class="flashcard">? × ?</div>
        <div class="input-area">
            <input type="number" id="answer-input" placeholder="?" min="0" autocomplete="off">
            <button id="check-btn" onclick="checkAnswer()">Check!</button>
        </div>
        <div id="feedback" class="feedback"></div>
    </div>

    <script>
        let score = 0, num1 = 0, num2 = 0, correctAnswer = 0, attempts = 0, timeLeft = 30, timerInterval;
        const input = document.getElementById('answer-input');

        function generateQuestion() {
            num1 = Math.floor(Math.random() * 12) + 1;
            num2 = Math.floor(Math.random() * 12) + 1;
            correctAnswer = num1 * num2;
            attempts = 0;
            timeLeft = 30;
            document.getElementById('flashcard').innerText = `${num1} × ${num2}`;
            input.value = '';
            input.disabled = false;
            document.getElementById('check-btn').disabled = false;
            input.focus();
            startTimer();
        }

        function startTimer() {
            clearInterval(timerInterval);
            updateTimerBar();
            timerInterval = setInterval(() => {
                timeLeft--;
                document.getElementById('time-text').innerText = timeLeft;
                updateTimerBar();
                if (timeLeft <= 0) {
                    clearInterval(timerInterval);
                    handleTimeout();
                }
            }, 1000);
        }

        function updateTimerBar() {
            const percentage = (timeLeft / 30) * 100;
            const bar = document.getElementById('timer-bar');
            bar.style.width = percentage + '%';
            bar.style.background = percentage > 50 ? '#4ADE80' : percentage > 20 ? '#FBBF24' : '#EF4444';
        }

        function checkAnswer() {
            let userAnswer = parseInt(input.value);
            let feedback = document.getElementById('feedback');
            if (isNaN(userAnswer)) return;

            if (userAnswer === correctAnswer) {
                clearInterval(timerInterval);
                let points = timeLeft > 15 ? 20 : 10;
                score += points;
                document.getElementById('score').innerText = score;
                feedback.className = "feedback correct-pop";
                feedback.innerHTML = `🎉 Correct! +${points} Points! 🌟`;
                input.disabled = true;
                setTimeout(generateQuestion, 2000);
            } else {
                attempts++;
                if (attempts < 3) {
                    feedback.className = "feedback wrong-pop";
                    feedback.innerHTML = `❌ Try again! (Attempt ${attempts}/3) 🤔`;
                    input.value = '';
                    input.focus();
                } else {
                    clearInterval(timerInterval);
                    feedback.className = "feedback wrong-pop";
                    feedback.innerHTML = `💡 Out of tries! ${num1} × ${num2} = ${correctAnswer}`;
                    input.disabled = true;
                    setTimeout(generateQuestion, 3500);
                }
            }
        }

        function handleTimeout() {
            let feedback = document.getElementById('feedback');
            feedback.className = "feedback wrong-pop";
            feedback.innerHTML = `⏰ Time's up! ${num1} × ${num2} = ${correctAnswer}`;
            input.disabled = true;
            setTimeout(generateQuestion, 3500);
        }

        input.addEventListener("keyup", function(event) {
            if (event.key === "Enter") checkAnswer();
        });

        generateQuestion();
    </script>
</body>
