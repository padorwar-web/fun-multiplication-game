    <script>
        let score = 0, num1 = 0, num2 = 0, correctAnswer = 0, attempts = 0, timeLeft = 30, timerInterval;
        let currentStreak = 0, topStreak = 0;
        const input = document.getElementById('answer-input');

        function generateQuestion() {
            num1 = Math.floor(Math.random() * 12) + 1;
            num2 = Math.floor(Math.random() * 12) + 1;
            correctAnswer = num1 * num2;
            attempts = 0;
            timeLeft = 30;
            document.getElementById('flashcard').innerText = `${num1} × ${num2}`;
            input.value = '';
            input.disabled = false;
            document.getElementById('check-btn').disabled = false;
            document.getElementById('feedback').className = "feedback";
            document.getElementById('feedback').innerHTML = "";
            input.focus();
            startTimer();
        }

        function startTimer() {
            clearInterval(timerInterval);
            updateTimerBar();
            timerInterval = setInterval(() => {
                timeLeft--;
                document.getElementById('time-text').innerText = timeLeft;
                updateTimerBar();
                if (timeLeft <= 0) {
                    clearInterval(timerInterval);
                    handleTimeout();
                }
            }, 1000);
        }

        function updateTimerBar() {
            const percentage = (timeLeft / 30) * 100;
            const bar = document.getElementById('timer-bar');
            bar.style.width = percentage + '%';
            bar.style.background = percentage > 50 ? '#4ade80' : percentage > 20 ? '#facc15' : '#f87171';
        }

        function checkAnswer() {
            let userAnswer = parseInt(input.value);
            let feedback = document.getElementById('feedback');
            if (isNaN(userAnswer)) return;

            if (userAnswer === correctAnswer) {
                clearInterval(timerInterval);
                let points = timeLeft > 15 ? 20 : 10;
                score += points;
                currentStreak++;
                
                if (currentStreak > topStreak) {
                    topStreak = currentStreak;
                }
                
                document.getElementById('score').innerText = score;
                document.getElementById('streak').innerText = currentStreak;
                document.getElementById('top-streak').innerText = topStreak;
                
                feedback.className = "feedback correct-pop";
                feedback.innerHTML = `🎉 Correct! +${points} Points! 🌟`;
                input.disabled = true;
                setTimeout(generateQuestion, 2000);
            } else {
                attempts++;
                if (attempts < 3) {
                    feedback.className = "feedback wrong-pop";
                    feedback.innerHTML = `❌ Try again! (Attempt ${attempts}/3) 🤔`;
